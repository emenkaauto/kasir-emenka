} finally { try { lock.releaseLock(); } catch (_) {} } } /** Upsert berbasis kolom pertama, dengan pembacaan sekali dan penulisan batch. */ function syncRows_(ss, sheetName, headers, rows) { let sheet = ss.getSheetByName(sheetName); Periksa baris pertama dan pulihkan header: ' + headers.join(' | ')); } // Pulihkan satu kasus kerusakan versi lama: header yang sama terduplikasi pada baris 2. if (lastRow >= 2) { const row2 = sheet.getRange(2, 1, 1, headers.length).getDisplayValues()[0].map(String); if (sameRow_(row2, headers)) sheet.deleteRow(2); } } const dataLastRow = sheet.getLastRow(); const existing = dataLastRow >= 2 ? sheet.getRange(2, 1, dataLastRow - 1, headers.length).getValues() : []; const indexByKey = new Map(); existing.forEach((row, i) => { const key = String(row[0] == null ? '' : row[0]).trim(); if (key) indexByKey.set(key, i); }); const updates = new Map(); const additions = []; (rows || []).forEach(rawRow => { const row = normalizeRow_(rawRow, headers.length); const key = String(row[0] == null ? '' : row[0]).trim(); if (!key) return; // Baris tanpa ID stabil tidak boleh ditambahkan berulang-ulang. const existingIndex = indexByKey.get(key); if (existingIndex !== undefined && existingIndex < existing.length) { existing[existingIndex] = row; updates.set(existingIndex + 2, row); } else if (existingIndex !== undefined) { // ID baru yang muncul lagi dalam payload yang sama: perbarui baris antrean, jangan duplikasi. additions[existingIndex - existing.length] = row; } else { const newIndex = existing.length + additions.length; indexByKey.set(key, newIndex); additions.push(row); } }); // Tulis update dalam rentang berurutan agar panggilan ke Spreadsheet Service minimal. const updateRows = Array.from(updates.keys()).sort((a, b) => a - b); if (updateRows.length) { let start = updateRows[0], group = [updates.get(start)], previous = start; for (let i = 1; i < updateRows.length; i++) { const rowNum = updateRows[i]; if (rowNum === previous + 1) group.push(updates.get(rowNum)); else { sheet.getRange(start, 1, group.length, headers.length).setValues(group); start = rowNum; group = [updates.get(rowNum)]; } previous = rowNum; } sheet.getRange(start, 1, group.length, headers.length).setValues(group); } if (additions.length) { sheet.getRange(sheet.getLastRow() + 1, 1, additions.length, headers.length).setValues(additions); } return { received: (rows || []).length, updated: updates.size, inserted: additions.length }; } function normalizeRow_(row, width) { const out = Array.isArray(row) ? row.slice(0, width) : []; while (out.length < width) out.push(''); return out.map(value => safeCell_(value == null ? '' : value)); } function validateSheetHeaders_(ss, sheetName, headers) { const sheet = ss.getSheetByName(sheetName); if (!sheet || sheet.getLastRow() === 0) return; const actual = sheet.getRange(1, 1, 1, headers.length).getDisplayValues()[0].map(String); if (!sameRow_(actual, headers)) { throw new Error('Header tab "' + sheetName + '" tidak cocok. Tidak ada data yang disinkronkan. Pulihkan baris pertama sesuai skema yang tercantum di fungsi doPost.'); } } // Mencegah input teks pengguna yang diawali karakter formula dijalankan sebagai formula Sheets. function safeCell_(value) { if (typeof value === 'string' && /^[=+@-]/.test(value)) return "'" + value; return value; } function sameRow_(a, b) { return a.length === b.length && a.every((v, i) => String(v == null ?
